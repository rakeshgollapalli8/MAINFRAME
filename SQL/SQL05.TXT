-- ********************************************************************
-- *** CREATE EMP_MASTER TABLE AND INDEX FOR PRIMARY KEY
-- ********************************************************************
   CREATE TABLE EMP_MASTER
       (
           EMP_NO     SMALLINT     NOT NULL     PRIMARY KEY,
           EMP_NAME   CHAR(10),
           BASIC_SAL  DECIMAL(7,2) NOT NULL
       ) IN OZA194DB.OZA194TS;

   CREATE UNIQUE INDEX IDX_M ON EMP_MASTER(EMP_NO);

-- ********************************************************************
-- *** CREATE EMP_TRANS TABLE AND INDEX FOR PRIMARY KEY
-- ********************************************************************
   CREATE TABLE EMP_TRANS
       (
           EMP_NO     SMALLINT     NOT NULL     PRIMARY KEY,
           BASIC_SAL  DECIMAL(7,2) NOT NULL,
           D_A        DECIMAL(7,2) NOT NULL,
           H_R_A      DECIMAL(7,2) NOT NULL,
           GROSS_SAL  DECIMAL(8,2) NOT NULL,
           ENTRY_DATE TIMESTAMP    NOT NULL
       ) IN OZA194DB.OZA194TS;

   CREATE UNIQUE INDEX IDX_T ON EMP_TRANS(EMP_NO);

   COMMIT;
