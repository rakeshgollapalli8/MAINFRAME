-- ********************************************************************
-- *** DROP TABLES
-- ********************************************************************
   DROP TABLE PROD_MASTER;
   DROP TABLE PROD_TRANS;
   COMMIT;

-- ********************************************************************
-- *** CREATE PROD_MASTER TABLE AND INDEX FOR PRIMARY KEY
-- ********************************************************************
   CREATE TABLE PROD_MASTER
      (
         PROD_NO       SMALLINT      NOT NULL     PRIMARY KEY,
         PROD_NAME     CHAR(10),
         RATE          DECIMAL(5,2)  NOT NULL,
         QTY           SMALLINT      NOT NULL
      ) IN OZA194DB.OZA194TS;

   CREATE UNIQUE INDEX IDX_PROD_M ON PROD_MASTER(PROD_NO);

-- ********************************************************************
-- *** CREATE PROD_TRANS TABLE AND INDEX FOR PRIMARY KEY
-- ********************************************************************
   CREATE TABLE PROD_TRANS
      (
         PROD_NO       SMALLINT      NOT NULL     PRIMARY KEY,
         RATE          DECIMAL(5,2)  NOT NULL,
         QTY           SMALLINT      NOT NULL,
         AMOUNT        DECIMAL(7,2)  NOT NULL,
         REMARKS       CHAR(25)      NOT NULL,
         ENTRY_DATE    CHAR(10)      NOT NULL
      ) IN OZA194DB.OZA194TS;

   CREATE UNIQUE INDEX IDX_PROD_T ON PROD_TRANS(PROD_NO);

   COMMIT;
