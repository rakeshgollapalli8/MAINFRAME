-- --- CREATE MASTER TABLE (DEPT) -------------------------------------
   CREATE TABLE DEPT
      (
          DEPT_NO        CHAR(03)      NOT NULL      PRIMARY KEY,
          DEPT_NAME      CHAR(10)      NOT NULL
      ) IN OZA194DB.OZA194TS;
   CREATE UNIQUE INDEX IDX_05 ON DEPT(DEPT_NO);

-- --- CREATE CHILD TABLE (EMP) ---------------------------------------
   CREATE TABLE EMP
      (
          EMP_NO         SMALLINT      NOT NULL      PRIMARY KEY,
          EMP_NAME       CHAR(10)      NOT NULL,
          DEPT_NO        CHAR(03),
          SALARY         DECIMAL(7,2)  NOT NULL,
          CONSTRAINT     CONST01       FOREIGN KEY(DEPT_NO)
                                       REFERENCES DEPT(DEPT_NO)
                                       ON DELETE SET NULL
      ) IN OZA194DB.OZA194TS;
   CREATE UNIQUE INDEX IDX_06 ON EMP(EMP_NO);
   COMMIT;
