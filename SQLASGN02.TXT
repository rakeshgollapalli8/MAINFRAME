-- --- DROP THE TABLE -------------------------------------------------
   DROP TABLE BANK_01;
   DROP TABLE BANK_02;
   COMMIT;

-- --- CREATE BANK01 TABLE AND INDEX FOR PRIMARY KEY ----------------
   CREATE TABLE BANK_01
       (
            CUST_NAME      CHAR(10)      NOT NULL     PRIMARY KEY,
            AC_OPEN_DATE   DATE          NOT NULL,
            BALANCE        DECIMAL(8,2)  NOT NULL
       ) IN OZA194DB.OZA194TS;
   CREATE UNIQUE INDEX IDX_33 ON BANK_01(CUST_NAME);

-- --- CREATE BANK02 TABLE AND INDEX FOR PRIMARY KEY ---------------
   CREATE TABLE BANK_02
       (
            CUST_NAME      CHAR(10)      NOT NULL     PRIMARY KEY,
            AC_OPEN_DATE   DATE          NOT NULL,
            BALANCE        DECIMAL(8,2)  NOT NULL
       ) IN OZA194DB.OZA194TS;
   CREATE UNIQUE INDEX IDX_34 ON BANK_02(CUST_NAME);
   COMMIT;

-- --- DISPLAY ROWS FOR PAN TABLE AND MOBILE TABLE --------------------
   SELECT * FROM BANK_01;
   SELECT * FROM BANK_02;

-- --- INNER JOIN -----------------------------------------------------
   SELECT BANK_01.CUST_NAME, BANK_01.AC_OPEN_DATE, BANK_01.BALANCE,
          BANK_02.CUST_NAME, BANK_02.AC_OPEN_DATE, BANK_02.BALANCE
     FROM BANK_01  INNER JOIN BANK_02
     ON   BANK_01.CUST_NAME = BANK_02.CUST_NAME;

-- --- LEFT OUTER JOIN ------------------------------------------------
   SELECT B1.CUST_NAME, B1.AC_OPEN_DATE, B1.BALANCE,
          B2.CUST_NAME, B2.AC_OPEN_DATE, B2.BALANCE
     FROM BANK_01 B1  LEFT OUTER JOIN  BANK_02 B2
     ON     B1.CUST_NAME = B2.CUST_NAME;

-- --- RIGHT OUTER JOIN -----------------------------------------------
   SELECT B2.CUST_NAME, B2.AC_OPEN_DATE, B2.BALANCE,
          B1.CUST_NAME, B1.AC_OPEN_DATE, B1.BALANCE
     FROM  BANK_02  B2  RIGHT OUTER JOIN  BANK_01 B1
     ON     B2.CUST_NAME = B1.CUST_NAME;
