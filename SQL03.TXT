-- --- DROP THE TABLE -------------------------------------------------
   DROP TABLE PAN_TABLE;
   DROP TABLE MOBILE_TABLE;
   COMMIT;

-- --- CREATE PAN TABLE AND INDEX FOR PRIMARY KEY ---------------------
   CREATE TABLE PAN_TABLE
       (
            EMP_NO       INT         NOT NULL     PRIMARY KEY,
            PAN_NO       CHAR(10)    NOT NULL
       ) IN OZA194DB.OZA194TS;
   CREATE UNIQUE INDEX IDX_03 ON PAN_TABLE(EMP_NO);

-- --- CREATE MOBILE TABLE AND INDEX FOR PRIMARY KEY ------------------
   CREATE TABLE MOBILE_TABLE
       (
            EMP_NO       INT         NOT NULL     PRIMARY KEY,
            MOBILE_NO    CHAR(10)    NOT NULL
       ) IN OZA194DB.OZA194TS;
   CREATE UNIQUE INDEX IDX_04 ON MOBILE_TABLE(EMP_NO);
   COMMIT;

-- --- DISPLAY ROWS FOR PAN TABLE AND MOBILE TABLE --------------------
   SELECT * FROM PAN_TABLE;
   SELECT * FROM MOBILE_TABLE;

-- --- INNER JOIN -----------------------------------------------------
   SELECT PAN_TABLE.EMP_NO, PAN_TABLE.PAN_NO,
          MOBILE_TABLE.EMP_NO, MOBILE_TABLE.MOBILE_NO
     FROM PAN_TABLE    INNER JOIN MOBILE_TABLE
     ON     PAN_TABLE.EMP_NO = MOBILE_TABLE.EMP_NO;

-- --- LEFT OUTER JOIN ------------------------------------------------
   SELECT P.EMP_NO, P.PAN_NO,
          M.EMP_NO, M.MOBILE_NO
     FROM  PAN_TABLE  P   LEFT OUTER JOIN  MOBILE_TABLE M
     ON     P.EMP_NO = M.EMP_NO;

-- --- RIGHT OUTER JOIN -----------------------------------------------
   SELECT PAN_TABLE.EMP_NO, PAN_TABLE.PAN_NO,
          MOBILE_TABLE.EMP_NO, MOBILE_TABLE.MOBILE_NO
     FROM  PAN_TABLE    RIGHT OUTER JOIN  MOBILE_TABLE
     ON     PAN_TABLE.EMP_NO = MOBILE_TABLE.EMP_NO;

-- --- FULL OUTER JOIN ------------------------------------------------
   SELECT PAN_TABLE.EMP_NO, PAN_TABLE.PAN_NO,
          MOBILE_TABLE.EMP_NO, MOBILE_TABLE.MOBILE_NO
     FROM  PAN_TABLE    FULL OUTER JOIN  MOBILE_TABLE
     ON     PAN_TABLE.EMP_NO = MOBILE_TABLE.EMP_NO;
